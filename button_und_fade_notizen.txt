2. NEUE DATEI: content/Button.tl
================================================================

Die Button-Logik liegt an einer Stelle, Menü und main.tl nutzen dieselbe
Klasse. Das Klicken läuft in update, draw zeichnet nur noch.

----------------------------------------------------------------
local record Button
   x: number
   y: number
   w: number
   h: number
   text: string
   fn: function()
   font: love.graphics.Font
   wasDown: boolean
end

local mt: metatable<Button> = { __index = Button }

function Button.new(x: number, y: number, w: number, h: number, text: string, fn: function(), font?: love.graphics.Font): Button
   local self: Button = {
      x = x, y = y, w = w, h = h,
      text = text,
      fn = fn,
      font = font or love.graphics.getFont(),
      wasDown = love.mouse.isDown(1),
   }
   return setmetatable(self, mt)
end

function Button:isHot(): boolean
   local mx, my = love.mouse.getPosition()
   return mx > self.x and mx < self.x + self.w
      and my > self.y and my < self.y + self.h
end

function Button:update()
   local isDown = love.mouse.isDown(1)
   if isDown and not self.wasDown and self:isHot() then
      self.fn()
   end
   self.wasDown = isDown
end

function Button:draw()
   if self:isHot() then
      love.graphics.setColor(0.45, 0.57, 0.33, 1.0)
   else
      love.graphics.setColor(0.4, 0.4, 0.5, 1.0)
   end
   love.graphics.rectangle("fill", self.x, self.y, self.w, self.h)

   love.graphics.setColor(0, 0, 0, 1)
   local textX = self.x + (self.w - self.font:getWidth(self.text)) * 0.5
   local textY = self.y + (self.h - self.font:getHeight()) * 0.5
   love.graphics.print(self.text, self.font, textX, textY)
   love.graphics.setColor(1, 1, 1, 1)
end

return Button

Die Zahl 100 ist ein Platzhalter. Für ein mittiges Menü muss die
Gesamthöhe aller Buttons eingerechnet werden.

----------------------------------------------------------------
local Button = require("content.Button")

local MainMenu = {}
MainMenu.__index = MainMenu

local BUTTON_HEIGHT: number = 64
local MARGIN: number = 16

local font: love.graphics.Font = nil
local buttons: {Button} = {}

local function add(text: string, fn: function())
   local w = love.graphics.getWidth() / 3
   local x = (love.graphics.getWidth() - w) * 0.5
   local y = 100 + #buttons * (BUTTON_HEIGHT + MARGIN)
   table.insert(buttons, Button.new(x, y, w, BUTTON_HEIGHT, text, fn, font))
end

function MainMenu.init()
   font = love.graphics.newFont(32)
   add("Start", function() print("Starting Evolution App") end)
   add("Load App", function() print("loading App files") end)
   add("Settings", function() print("Settings Button") end)
   add("Credits", function() print("Credits Button") end)
   add("Exit", function() love.event.quit(0) end)
end

function MainMenu.update()
   for _, b in ipairs(buttons) do b:update() end
end

function MainMenu.draw()
   for _, b in ipairs(buttons) do b:draw() end
end

return MainMenu

5. ANIMATION TIMER WIE IN JAVAFX
================================================================

Variante 1: love.update(dt)
Entspricht direkt dem AnimationTimer. Läuft wie handle(long now) einmal
pro Frame. dt sind die Sekunden seit dem letzten Frame, nicht Nanosekunden.
Man muss nichts starten, die Funktion läuft immer.

----------------------------------------------------------------
local alpha: number = 1
local fadeSpeed: number = 1 / 0.3   -- in 0.3 Sekunden ausblenden
local isFading: boolean = false

function love.update(dt: number)
   if isFading then
      alpha = math.max(0, alpha - fadeSpeed * dt)
      if alpha == 0 then
         isFading = false
         isNNVisible = false
      end
   end
end
----------------------------------------------------------------

Entspricht ungefähr FadeTransition: Dauer und Zielwert angeben, am Ende
